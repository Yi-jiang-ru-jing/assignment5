cmake_minimum_required(VERSION 3.8)
project(mid360_mapping)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(PCL REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(mapping_node src/main.cpp src/mapping_node.cpp)
add_executable(map_processor_node src/map_processor_node.cpp)
foreach(target mapping_node map_processor_node)
  target_include_directories(${target} PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(${target} ${PCL_LIBRARIES})
  ament_target_dependencies(${target} rclcpp rcl_interfaces sensor_msgs nav_msgs geometry_msgs std_srvs livox_ros_driver2 pcl_conversions tf2 tf2_ros tf2_geometry_msgs tf2_eigen)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
install(TARGETS mapping_node map_processor_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
ament_package()
